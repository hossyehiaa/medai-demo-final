    84→// ── F13: Shareable Link ────────────────────────────────────────────
    85→function generateShareLink(messageId: string, content: string): string {
    86→  const data = btoa(encodeURIComponent(content.substring(0, 500)));
    87→  return `${typeof window !== 'undefined' ? window.location.origin : ''}/share?id=${messageId}&d=${data}`;
    88→}

   100→  const [showScrollButton, setShowScrollButton] = useState(false);

   102→
   103→  // ── Feature State ────────────────────────────────────────────────
   104→  const [showCrisis, setShowCrisis] = useState(false);
   105→  const [activePipelineStage, setActivePipelineStage] = useState(-1);
   106→  const [currentEvidence, setCurrentEvidence] = useState<EvidenceChunk[]>([]);
   107→  const [currentMeta, setCurrentMeta] = useState<MetaEvent | null>(null);
   108→  const [mode, setMode] = useState<'clinician' | 'patient'>('clinician');

   115→  const [showCommandPalette, setShowCommandPalette] = useState(false);
   116→
   117→  // F1: Source Viewer
   118→  const [sourceViewer, setSourceViewer] = useState<{
   119→    isOpen: boolean;
   120→    docName: string;
   121→    page?: number;
   122→    quote?: string;

   124→
   125→  // F13: Shareable link
   126→  const [copiedLinkId, setCopiedLinkId] = useState<string | null>(null);

   175→        id: crypto.randomUUID(),
   176→        role: 'user',
   177→        content: q,
   178→      };
   179→
   180→      const assistantId = crypto.randomUUID();

   182→        id: assistantId,
   183→        role: 'assistant',
   184→        content: '',
   185→        status: 'OK',
   186→        isStreaming: true,
   187→        evidence: [],
   188→        citations: [],
   189→        stages: [],

   219→              ),
   220→            );
   221→          },

   223→          onStage: (stage) => {

   231→              ),
   232→            );
   233→          },

   235→          onEvidence: (chunks) => {
   236→            setCurrentEvidence(chunks);
   237→            setMessages((prev) =>
   238→              prev.map((msg) =>
   239→                msg.id === assistantId ? { ...msg, evidence: chunks } : msg,
   240→              ),
   241→            );
   242→          },
   243→
   244→          onMeta: (meta) => {
   245→            setCurrentMeta(meta);
   246→            setMessages((prev) =>
   247→              prev.map((msg) =>
   248→                msg.id === assistantId ? { ...msg, meta } : msg,
   249→              ),
   250→            );
   251→          },
   252→
   253→          onCitations: (citations) => {
   254→            setMessages((prev) =>
   255→              prev.map((msg) =>
   256→                msg.id === assistantId ? { ...msg, citations } : msg,
   257→              ),
   258→            );
   259→          },
   260→
   261→          onComplete: (metadata) => {
   262→            const isCrisis = metadata.referral988 || metadata.status === 'CRISIS';
   263→            setMessages((prev) =>
   264→              prev.map((msg) =>
   265→                msg.id === assistantId
   266→                  ? {
   267→                      ...msg,
   268→                      isStreaming: false,
   269→                      status:
   270→                        (metadata.status as ExtendedChatMessageData['status']) ||
   271→                        'OK',
   272→                      content: msg.content || metadata.message || '',
   273→                      referral988: isCrisis,
   274→                    }
   275→                  : msg,
   276→              ),
   277→            );
   278→            setIsStreaming(false);
   279→            setActivePipelineStage(-1);
   280→
   281→            // Show crisis screen for crisis queries
   282→            if (isCrisis) {
   283→              setShowCrisis(true);
   284→            }
   285→

   287→            const lastMessages = messages.slice(-4);
   288→            saveSession({
   289→              id: chatId,
   290→              title: q.substring(0, 60),
   291→              timestamp: Date.now(),
   292→              messageCount: lastMessages.length + 2,
   293→              preview: q.substring(0, 80),

   296→
   297→          onError: (error) => {

   299→              prev.map((msg) =>
   300→                msg.id === assistantId
   301→                  ? {
   302→                      ...msg,
   303→                      isStreaming: false,
   304→                      content:
   305→                        msg.content +
   306→                        `\n\n⚠️ Error: ${error.message}. Please try again.`,
   307→                      status: 'FAILED_OUTPUT',
   308→                    }
   309→                  : msg,
   310→              ),
   311→            );
   312→            setIsStreaming(false);
   313→            setActivePipelineStage(-1);
   314→          },
   315→        },
   316→        controller.signal,
   317→        locale,
   318→      );
   319→    },
   320→    [query, isStreaming, locale, chatId, messages, activePipelineStage],
   321→  );

   382→            onClick={() => setShowHistory(true)}

   384→            aria-label={t('chat.sessionHistory')}

   386→            <History className="w-4 h-4" />

   389→          {/* Brand Logo (F14: ECG) */}
   390→          <BrandLogo variant="ecg" size={36} />

   393→        <div className="flex items-center gap-2">

   396→            <Shield className="w-3 h-3" />
   397→            Safety-Gated

   399→
   400→          {/* F10: Clinician/Patient Mode */}
   401→          <ClinicianPatientToggle mode={mode} onToggle={setMode} />

   403→          {/* F6: Language Toggle */}
   404→          <LanguageToggle />
   405→
   406→          {/* F12: Command Palette trigger */}
   407→          <button
   408→            onClick={() => setShowCommandPalette(true)}

   411→            <Command className="w-3 h-3" />

   435→          <div className="max-w-3xl mx-auto px-4 sm:px-6 py-6">
   436→            {/* ── Welcome Hero ── */}
   437→            {!hasMessages && !showScreener && (
   438→              <motion.div
   439→                className="flex flex-col items-center justify-center min-h-[60vh] text-center"

   543→                {messages.map((msg) => (

   545→                    <ChatMessage message={msg} />
   546→
   547→                    {/* Post-message features for assistant messages */}
   548→                    {msg.role === 'assistant' && !msg.isStreaming && msg.content && (

   550→                        className="mt-3 ml-11 space-y-3"

   563→                        {/* F4: Quality HUD + F2: Pipeline Trace */}

   585→                            {/* F13: Shareable link */}
   586→                            <motion.button
   587→                              onClick={() => handleShare(msg.id, msg.content)}
   588→                              className="flex items-center gap-1.5 px-3 py-1.5 bg-[#131B3F] border border-[#223058] rounded-lg text-xs text-[#AAB4D4] hover:text-[#F5F7FB] hover:border-[#4ECDC4]/30 transition-colors"
   589→                              whileTap={{ scale: 0.95 }}
   590→                              aria-label={t('chat.copyLink')}

   592→                              {copiedLinkId === msg.id ? (
   593→                                <>
   594→                                  <Check className="w-3.5 h-3.5 text-[#4ECDC4]" />
   595→                                  <span className="text-[#4ECDC4]">{t('chat.copied')}</span>

   599→                                  <Share2 className="w-3.5 h-3.5" />

   602→                              )}
   603→                            </motion.button>
   604→                          </div>
   605→                        </div>
   606→                      </motion.div>
   607→                    )}
   608→                  </div>
   609→                ))}
   610→
   611→                {/* Streaming caret */}
   612→                {isStreaming &&
   613→                  messages[messages.length - 1]?.role === 'assistant' && (
   614→                    <div className="flex items-start gap-3">
   615→                      <div className="w-8" />
   616→                      <div className="max-w-[85%] sm:max-w-[75%] px-4 py-1">

   618→                      </div>
   619→                    </div>
   620→                  )}
   621→
   622→                {/* Typing indicator */}
   623→                {isStreaming && messages[messages.length - 1]?.content === '' && (
   624→                  <TypingIndicator />
   625→                )}
   626→
   627→                {/* Active pipeline trace while streaming */}
   628→                {isStreaming && activePipelineStage >= 0 && (
   629→                  <div className="ml-11">
   630→                    <PipelineTrace activeStage={activePipelineStage} />
   631→                  </div>
   632→                )}
   633→
   634→                {/* Active evidence while streaming */}
   635→                {isStreaming && currentEvidence.length > 0 && (
   636→                  <div className="ml-11">
   637→                    <EvidencePanel
   638→                      chunks={currentEvidence}
   639→                      onViewSource={(doc, page) => handleViewSource(doc, page)}
   640→                    />
   641→                  </div>
   642→                )}
   643→              </div>
   644→            )}
   645→
   646→            <div ref={messagesEndRef} />
   647→          </div>
   648→        </div>
   649→
   650→        {/* ── Input Bar ── */}
   651→        <div className="sticky bottom-0 z-10 px-4 sm:px-6 py-3 bg-[#0A1128]/80 backdrop-blur-xl border-t border-[#223058]">
   652→          <div className="max-w-3xl mx-auto flex items-end gap-2">
   653→            {/* PHQ-9 quick launch */}
   654→            <button
   655→              onClick={() => setShowScreener(true)}
   656→              className="shrink-0 p-2.5 rounded-xl bg-[#131B3F] border border-[#223058] text-[#AAB4D4] hover:text-[#4ECDC4] hover:border-[#4ECDC4]/30 transition-colors"
   657→              aria-label={t('phq9.title')}
   658→              title={t('phq9.title')}
   659→            >
   660→              <ClipboardList className="w-4 h-4" />
   661→            </button>
   662→
   663→            <div className="flex-1 relative">
   664→              <textarea
   665→                ref={inputRef}
   666→                value={query}
   667→                onChange={handleInput}
   668→                onKeyDown={handleKeyDown}
   669→                placeholder={t('chat.placeholder')}
   670→                rows={1}
   671→                className="
   672→                  w-full resize-none rounded-2xl px-4 py-3 pr-4
   673→                  bg-[#131B3F] border border-[#223058]
   674→                  focus:border-[#4ECDC4]/40 focus:ring-2 focus:ring-[#4ECDC4]/10
   675→                  placeholder:text-[#7683AB]
   676→                  text-sm leading-relaxed text-[#F5F7FB]
   677→                  transition-all duration-200
   678→                  outline-none
   679→                "
   680→                style={{ maxHeight: '120px' }}
   681→                disabled={isStreaming}
   682→                aria-label="Type your clinical question"
   683→                id="chat-input"
   684→              />
   685→            </div>
   686→            <SendButton
   687→              onClick={() => handleSend()}
   688→              disabled={!query.trim() || isStreaming}
   689→              loading={isStreaming}
   690→            />
   691→          </div>
   692→
   693→          {/* New chat button */}
   694→          {hasMessages && (
   695→            <div className="max-w-3xl mx-auto mt-2 flex justify-center">
   696→              <button
   697→                onClick={handleNewChat}
   698→                className="text-xs text-[#7683AB] hover:text-[#AAB4D4] transition-colors cursor-pointer"
   699→              >
   700→                {t('chat.newChat')}
   701→              </button>
   702→            </div>
   703→          )}
   704→        </div>
   705→      </main>
   706→
   707→      {/* ── Footer ── */}
   708→      <footer className="px-4 py-3 text-center border-t border-[#223058]/50">
   709→        <div className="flex items-center justify-center gap-4 text-[10px] text-[#7683AB]">
   710→          <span>{t('footer.crisisLine')}</span>
   711→          {locale === 'ar' && <span>{t('footer.egyptLine')}</span>}
   712→          <span>|</span>
   713→          <span>{t('footer.madeInEgypt')}</span>
   714→        </div>
   715→        <p className="text-[10px] text-[#7683AB]/60 mt-1 max-w-2xl mx-auto leading-relaxed">
   716→          {t('footer.disclaimer')}
   717→        </p>
   718→      </footer>
   719→
   720→      {/* ── Overlays ── */}
   721→
   722→      {/* F5: Crisis Screen */}
   723→      <CrisisScreen isVisible={showCrisis} onReturn={() => setShowCrisis(false)} />
   724→
   725→      {/* F1: Source Viewer Drawer */}
   726→      <SourceViewer
   727→        isOpen={sourceViewer.isOpen}
   728→        onClose={() => setSourceViewer({ isOpen: false, docName: '' })}
   729→        docName={sourceViewer.docName}
   730→        page={sourceViewer.page}
   731→        quote={sourceViewer.quote}
   732→      />
   733→
   734→      {/* F11: History Sidebar */}
   735→      <HistorySidebar
   736→        isOpen={showHistory}
   737→        onClose={() => setShowHistory(false)}
   738→        onNewChat={handleNewChat}
   739→        onLoadSession={() => {
   740→          setShowHistory(false);
   741→        }}
   742→      />
   743→
   744→      {/* F12: Command Palette */}
   745→      <CommandPalette
   746→        isOpen={showCommandPalette}
   747→        onClose={() => setShowCommandPalette(false)}
   748→        onNewChat={handleNewChat}
   749→        onExport={() => {}}
   750→        onScreener={() => setShowScreener(true)}
   751→      />
   752→
   753→      {/* Scroll to latest */}
   754→      <ScrollToLatest visible={showScrollButton} onClick={scrollToBottom} />
   755→
   756→      {/* ── PHQ-9 Modal (when chat has messages) ── */}
   757→      <AnimatePresence>
   758→        {showScreener && hasMessages && (
   759→          <motion.div
   760→            className="fixed inset-0 z-50 flex items-center justify-center p-4"
   761→            initial={{ opacity: 0 }}
   762→            animate={{ opacity: 1 }}
   763→            exit={{ opacity: 0 }}
   764→          >
   765→            <div
   766→              className="absolute inset-0 bg-black/50"
   767→              onClick={() => setShowScreener(false)}
   768→            />
   769→            <motion.div
   770→              className="relative max-w-xl w-full max-h-[80vh] overflow-y-auto navy-scroll"
   771→              initial={{ scale: 0.95, y: 20 }}
   772→              animate={{ scale: 1, y: 0 }}
   773→              exit={{ scale: 0.95, y: 20 }}
   774→              transition={{ stiffness: 120, damping: 14 }}
   775→            >
   776→              <button
   777→                onClick={() => setShowScreener(false)}
   778→                className="absolute top-4 right-4 z-10 p-2 rounded-full bg-[#131B3F] border border-[#223058] text-[#AAB4D4] hover:text-[#F5F7FB] transition-colors"
   779→              >
   780→                ✕
   781→              </button>
   782→              <PHQ9Screener />
   783→            </motion.div>
   784→          </motion.div>
   785→        )}
   786→      </AnimatePresence>
   787→    </div>
   788→  );
